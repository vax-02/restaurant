!define table(x) class x << (T,#FFAAAA) >>
!define primary_key(x) <b><u>x</u></b>
!define foreign_key(x) <i>x</i>

' ============================================
' TABLAS DEL SISTEMA
' ============================================

table(users) {
  + primary_key(id) : BIGINT UNSIGNED
  --
  name : VARCHAR(255)
  email : VARCHAR(255) UNIQUE
  email_verified_at : TIMESTAMP
  password : VARCHAR(255)
  role : ENUM('admin','ayudante')
  remember_token : VARCHAR(100)
  created_at : TIMESTAMP
  updated_at : TIMESTAMP
}

table(deliveries) {
  + primary_key(id) : BIGINT UNSIGNED
  --
  code : VARCHAR(50)
  status : TINYINT(1)
  name : VARCHAR(50)
  lastname : VARCHAR(50)
  cellphone : VARCHAR(8)
  user_telegram : VARCHAR(50)
  created_at : TIMESTAMP
  updated_at : TIMESTAMP
}

table(products) {
  + primary_key(id) : BIGINT UNSIGNED
  --
  name : VARCHAR(255)
  description : TEXT
  price : DECIMAL(8,2)
  category : ENUM('plato','bebida')
  image : VARCHAR(255)
  available : TINYINT(1)
  created_at : TIMESTAMP
  updated_at : TIMESTAMP
}

table(daily_availability) {
  + primary_key(id) : BIGINT UNSIGNED
  --
  foreign_key(product_id) : BIGINT UNSIGNED
  date : DATE
  stock : INT(11)
  created_at : TIMESTAMP
  updated_at : TIMESTAMP
}

table(buys) {
  + primary_key(id) : BIGINT UNSIGNED
  --
  comprobante : VARCHAR(255)
  foreign_key(delivery_id) : BIGINT UNSIGNED
  client : VARCHAR(200)
  type : ENUM('delivery','restaurant')
  status : ENUM('-1','0','1','2')
  cancel_reason : VARCHAR(255)
  latitude : DECIMAL(10,8)
  longitude : DECIMAL(11,8)
  created_at : TIMESTAMP
  updated_at : TIMESTAMP
}

table(buy_details) {
  + primary_key(id) : BIGINT UNSIGNED
  --
  foreign_key(buy_id) : BIGINT UNSIGNED
  foreign_key(product_id) : BIGINT UNSIGNED
  price : DECIMAL(8,2)
  created_at : TIMESTAMP
  updated_at : TIMESTAMP
}

table(restaurants) {
  + primary_key(id) : BIGINT UNSIGNED
  --
  qr : VARCHAR(255)
  created_at : TIMESTAMP
  updated_at : TIMESTAMP
}

table(sessions) {
  + primary_key(id) : VARCHAR(255)
  --
  foreign_key(user_id) : BIGINT UNSIGNED
  ip_address : VARCHAR(45)
  user_agent : TEXT
  payload : LONGTEXT
  last_activity : INT(11)
}

table(cache) {
  + primary_key(key) : VARCHAR(255)
  --
  value : MEDIUMTEXT
  expiration : BIGINT(20)
}

table(jobs) {
  + primary_key(id) : BIGINT UNSIGNED
  --
  queue : VARCHAR(255)
  payload : LONGTEXT
  attempts : SMALLINT UNSIGNED
  reserved_at : INT UNSIGNED
  available_at : INT UNSIGNED
  created_at : INT UNSIGNED
}

table(failed_jobs) {
  + primary_key(id) : BIGINT UNSIGNED
  --
  uuid : VARCHAR(255) UNIQUE
  connection : VARCHAR(255)
  queue : VARCHAR(255)
  payload : LONGTEXT
  exception : LONGTEXT
  failed_at : TIMESTAMP
}

table(password_reset_tokens) {
  + primary_key(email) : VARCHAR(255)
  --
  token : VARCHAR(255)
  created_at : TIMESTAMP
}

' ============================================
' RELACIONES
' ============================================

' Un pedido (buys) puede tener un delivery asignado
deliveries ||--o{ buys : "asigna"

' Un pedido tiene muchos detalles
buys ||--o{ buy_details : "contiene"

' Un producto aparece en muchos detalles
products ||--o{ buy_details : "incluido en"

' Un producto tiene disponibilidad por fecha
products ||--o{ daily_availability : "stock por fecha"

' Un usuario (admin/ayudante) tiene sesiones
users ||--o{ sessions : "tiene"

' ============================================
' NOTAS Y LEYENDA
' ============================================

note right of buys
  <b>Estados (status):</b>
  -1 = Cancelado
   0 = Pendiente
   1 = Confirmado/Aceptado
   2 = Entregado
end note

note right of users
  <b>Roles (role):</b>
  admin = Administrador
  ayudante = Asistente
end note

note right of products
  <b>Categorías (category):</b>
  plato = Plato principal
  bebida = Bebida
end note

note right of daily_availability
  <b>Clave única:</b>
  (product_id, date)
  Un producto solo tiene
  un stock por fecha
end note

@enduml

' ============================================
' CAPA DE PRESENTACIÓN (FRONTEND)
' ============================================

package "Capa de Presentación" {
  component "Bot de Telegram\n(Cliente / Delivery)" as BotClient
  component "Panel Web\n(Administrador / Ayudante)" as PanelWeb
}

' ============================================
' CAPA DE NEGOCIO (BACKEND)
' ============================================

package "Capa de Negocio" {
  
  component "API REST\n(Laravel)" as API {
    portin "HTTP Requests" as HTTPIn
    portout "JSON Responses" as JSONOut
  }
  
  component "Servicio del Bot\n(Telegram Bot Service)" as BotService {
    portin "Webhook / Polling" as WebhookIn
    portout "Respuestas" as BotOut
  }
  
  component "Servicio de Notificaciones" as NotifService {
    portin "Eventos de estado" as EventIn
    portout "Notificaciones" as NotifOut
  }
  
  component "Servicio de QR" as QRService {
    portin "Solicitud QR" as QRIn
    portout "Imagen QR" as QROut
  }
}

' ============================================
' CAPA DE DATOS
' ============================================

package "Capa de Datos" {
  database "MySQL / MariaDB\n(Base de Datos)" as DB {
    rectangle "Tablas Principales" {
      card "users" as T_Users
      card "deliveries" as T_Deliveries
      card "products" as T_Products
      card "daily_availability" as T_Stock
      card "buys" as T_Buys
      card "buy_details" as T_Details
    }
  }
  
  file "Almacenamiento\n(Comprobantes)" as Storage
}

' ============================================
' SERVICIOS EXTERNOS
' ============================================

package "Servicios Externos" {
  component "Telegram Bot API" as TelegramAPI
  component "API de QR\n(qrserver.com)" as QRAPI
}

' ============================================
' CONEXIONES - CLIENTES
' ============================================

BotClient --> TelegramAPI : "1. Mensajes / Comandos"
TelegramAPI --> BotService : "2. Webhook / Polling"

' ============================================
' CONEXIONES - PANEL WEB
' ============================================

PanelWeb --> API : "3. HTTP (AJAX / Formularios)"
API --> BotService : "4. Eventos (crear pedido,\ncambiar estado)"

' ============================================
' CONEXIONES - SERVICIOS INTERNOS
' ============================================

BotService --> DB : "5. SQL (CRUD)"
API --> DB : "6. SQL (CRUD)"

BotService --> NotifService : "7. Evento de cambio de estado"
NotifService --> TelegramAPI : "8. sendMessage / sendPhoto"

BotService --> QRService : "9. Solicitar QR"
QRService --> QRAPI : "10. Generar QR"

BotService --> Storage : "11. Guardar comprobantes"
API --> Storage : "12. Leer comprobantes"

' ============================================
' CONEXIONES - BASE DE DATOS (DETALLE)
' ============================================

T_Users --> T_Buys : "FK (cliente)"
T_Deliveries --> T_Buys : "FK (delivery)"
T_Products --> T_Stock : "FK (product_id)"
T_Products --> T_Details : "FK (product_id)"
T_Buys --> T_Details : "FK (buy_id)"
